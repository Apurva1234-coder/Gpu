cmake_minimum_required(VERSION 3.16)
project(sovereign_presolve LANGUAGES CXX)
option(SOVEREIGN_ENABLE_CUDA "Build optional CUDA backend" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(sovereign_presolve INTERFACE)
target_include_directories(sovereign_presolve INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(SOVEREIGN_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(sovereign_cuda STATIC cuda/CudaBackend.cu)
  target_compile_definitions(sovereign_cuda PUBLIC SOVEREIGN_HAS_CUDA=1)
  target_link_libraries(sovereign_cuda PUBLIC CUDA::cudart CUDA::cublas CUDA::cusparse)
  target_link_libraries(sovereign_presolve INTERFACE sovereign_cuda)
else()
  add_library(sovereign_cuda STATIC cuda/CudaBackend.cpp)
  target_include_directories(sovereign_cuda PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_link_libraries(sovereign_presolve INTERFACE sovereign_cuda)
endif()
add_executable(sovereign_presolve_cli src/main.cpp)
target_link_libraries(sovereign_presolve_cli PRIVATE sovereign_presolve)
enable_testing()
add_executable(sovereign_presolve_tests tests/presolve_tests.cpp)
target_link_libraries(sovereign_presolve_tests PRIVATE sovereign_presolve)
add_test(NAME presolve_tests COMMAND sovereign_presolve_tests)
add_executable(sovereign_cutting_plane_tests tests/cutting_plane_tests.cpp)
target_link_libraries(sovereign_cutting_plane_tests PRIVATE sovereign_presolve)
add_test(NAME cutting_plane_tests COMMAND sovereign_cutting_plane_tests)
add_executable(sovereign_lp_relaxation_tests tests/lp_relaxation_tests.cpp)
target_link_libraries(sovereign_lp_relaxation_tests PRIVATE sovereign_presolve)
add_test(NAME lp_relaxation_tests COMMAND sovereign_lp_relaxation_tests)
add_executable(sovereign_feasibility_pump_tests tests/feasibility_pump_tests.cpp)
target_link_libraries(sovereign_feasibility_pump_tests PRIVATE sovereign_presolve)
add_test(NAME feasibility_pump_tests COMMAND sovereign_feasibility_pump_tests)
add_executable(sovereign_nla_tests tests/nla_tests.cpp)
target_link_libraries(sovereign_nla_tests PRIVATE sovereign_presolve)
add_test(NAME nla_tests COMMAND sovereign_nla_tests)
add_executable(sovereign_ipm_tests tests/ipm_tests.cpp)
target_link_libraries(sovereign_ipm_tests PRIVATE sovereign_presolve)
add_test(NAME ipm_tests COMMAND sovereign_ipm_tests)
